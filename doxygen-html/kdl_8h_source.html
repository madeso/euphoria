66 // Cast the number to a fundamental arithmetic type (no bounds checking,
67 // no support for strings)
68 template <arithmetic_concept T>

70 {
71 if (std::holds_alternative<integer_type>(m_value)) {

73 } else if (std::holds_alternative<float_type>(m_value)) {
